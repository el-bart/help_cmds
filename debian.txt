# add 32-bit architecture to a supported list
dpkg --add-architecture i386
apt-get update

# list obsolete packjages
apt list '~o'
# remove obsolete packages
apt purge '~o'

# list removed but not purged packages
apt list '~c'
# purge removed but not purged packages
apt pruge '~c'

# don't mount /tmp/ in RAM disk (default since debian:13)
systemctl mask tmp.mount
touch /etc/tmpfiles.d/tmp.conf

# apt install ...
# usefull packages:
apt-listchanges
apt-file
debian-security-support  # provides: check-support-status
fwupd # provides: fwupdmgr
smartmontools # provides: smartctl

# list packages from outside of the core distro
apt list '?narrow(?installed, ?not(?origin(Debian)))'

# install package w/o recommended part
apt install --no-install-recommends some-package

# update source files format (debian 12 -> 13)
apt modernize-sources

# make OS updated in non-interactive mode
export DEBIAN_FRONTEND=noninteractive
apt-get dist-upgrade -y


# download given package into a current directory (no need for root)
apt download htop

# to list dependencies of a package:
apt rdepends --recurse htop
# or as a list:
apt rdepends --recurse htop 2>/dev/null | grep '^  Depends' | awk '{ print $2 }' | sort -u

# to download the pacakge (with deps), w/o the ones already installed:
apt-get --download-only -y install htop
cp -v /var/cache/apt/archives/*.deb /path/to/destination/
