#
# curl
#

# set of curl flags that will not be verbose but still will show errors and return error codes back to shell
curl \
  --silent \
  --show-error \
  --fail \
  --fail-early \
  "https://foo/bar/net"

# the same, but shorter:
curl -sSf --fail-early http://foo.bar/stuff
# follow all redirections
curl -L baszerr.eu

# common use case to make curl useful in scripts:
curl -fsSL baszerr.eu


#
# wget
#

# get page ignoring certs checks:
wget --no-check-certificate https://baszerr.eu

# get page and print on the screen:
wget -O - https://baszerr.eu

# get page showing dotted progress bar
wget --progress=dot https://baszerr.eu
